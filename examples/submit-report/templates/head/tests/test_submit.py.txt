from unittest.mock import AsyncMock, patch

import pytest

from app import submit_report


@pytest.mark.asyncio
async def test_submit_returns_success():
    with (
        patch("app.submit_to_provider", new_callable=AsyncMock, return_value={"success": True}),
        patch("app.build_notification", new_callable=AsyncMock, return_value="summary"),
        patch("app.send_slack", new_callable=AsyncMock),
    ):
        result = await submit_report("report", approved=True)
        assert result["success"] is True
